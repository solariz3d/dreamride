# Dark half-time: sixteen bars of our own, E harmonic minor, 144 bpm felt at 72.
#
# The contrast to citypop_v2: straight time, almost no harmony, weight in the low end.
#   808       the bass is the kick's low end: a driven sine that punches down from above at each
#             hit and SLIDES up an octave on legato steps, without re-striking.
#   half-time the snare lands once a bar, on beat 3; the kick follows the 808's hits.
#   hats      a rate function: each beat plays 8ths, 16ths, triplets or 32nds, and the rolls ramp
#             in velocity, at the ends of phrases.
#   hook      one two-bar bell motif in a 3-3-2 rhythm, the harmonic minor's raised seventh (D#)
#             giving it its edge, repeated over every chord, its last note lifted at phrase ends.
#   form      the motif alone, filtered, opening (1-4) -> a roll into the drop (5-12) -> drums out,
#             808 held (13-14) -> back in (15) -> the 808 drops out for the last beat (16), so the
#             loop's first downbeat lands on silence and air.
#
# Pitch is in semitones from A4 = 440 Hz. 144 bpm makes one beat exactly 20000 samples.

bpm 144
beats 64
rate 48000

# ======================================================================== time

step(x) = floor(4x)                         # straight sixteenths: no swing in this style
ph(x) = frac(4x)
bar(s) = mod(floor(s / 16), 16)
pos(s) = mod(s, 16)
d16(n, i) = mod(floor(n / 10^(15 - i)), 10)
smoothstep(u) = u^2(3 - 2u)
on(n, s) = d16(n, bar(s))

# ======================================================================== harmony

key = -5                                    # E4
# The harmonic minor scale: 0 2 3 5 7 8 11 (the 11 does not fit a digit, so it is spelled out).
hmi(i) = if(i == 6, 11, mod(floor(235780 / 10^(6 - i)), 10))
hm(d) = 12 floor(d / 7) + hmi(mod(d, 7))

# Four chords, two bars each, twice: i (Em)  bVI (C)  iv (Am)  V (B, with the raised D#).
chord(s) = mod(floor(s / 32), 4)
croot(s) = d16(857, chord(s) + 12)          # 0 8 5 7
cthird(s) = d16(3443, chord(s) + 12)        # minor, major, minor, major

# Three triad tones folded into one octave and sorted (lowest, middle, highest).
wrap(z, lo) = lo + mod(z - lo, 12)
ta(s, lo) = wrap(croot(s), lo)
tb(s, lo) = wrap(croot(s) + cthird(s), lo)
tc(s, lo) = wrap(croot(s) + 7, lo)
tlow(s, lo) = min(min(ta(s, lo), tb(s, lo)), tc(s, lo))
thigh(s, lo) = max(max(ta(s, lo), tb(s, lo)), tc(s, lo))
tri(k, s, lo) = if(k < 1, tlow(s, lo), if(k < 2, ta(s, lo) + tb(s, lo) + tc(s, lo) - tlow(s, lo) - thigh(s, lo), thigh(s, lo)))

# ======================================================================== arrangement (a digit per bar)

drums_on = 0000111111110011                 # the motif alone, then the drop; out for 13-14
hats_on = 0001111111111111                    # in from the roll at the end of bar 4, through the bridge
b808s = 0000121212123312                    # 1 main, 2 answer, 3 held (bridge), 4 drop-out bar
snares = 0000111211120011                   # 2 adds a pickup at the end of the phrase

# ======================================================================== 808

# Codes: 0 rest, 9 hold, 1 root, 3 fifth, 6 slide up an octave from the note before (legato).
b808d(s) = d16(if(on(b808s, s) == 0, 0, if(bar(s) == 15, 1999999900190000, if(on(b808s, s) == 1, 1999999900199969, if(on(b808s, s) == 2, 1999990199039910, 1999999919999999)))), pos(s))
held8(s) = (b808d(s) == 9)(1 + (b808d(s-1) == 9)(1 + (b808d(s-2) == 9)(1 + (b808d(s-3) == 9)(1 + (b808d(s-4) == 9)(1 + (b808d(s-5) == 9)(1 + (b808d(s-6) == 9)))))))
onset8(s) = s - held8(s)
note8(c, s) = croot(s) + if(c == 3, 7, if(c == 6, 12, 0))
# The pitch: the note, plus a punch that falls an octave-ish to it over a third of a sixteenth
# at every struck hit, or a glide up from the note before on a slide.
age8(x) = held8(step(x)) + ph(x)
punch(x) = 7 (1 - smoothstep(clamp(age8(x) / 0.35, 0, 1)))
slide(x) = -12 (1 - smoothstep(clamp(age8(x) / 0.6, 0, 1)))
p808(x) = note8(b808d(onset8(step(x))), onset8(step(x))) + if(b808d(onset8(step(x))) == 6, slide(x), punch(x))
# Legato into holds and slides; anything else re-strikes.
gate808(x) = if(b808d(step(x)) == 0, -1, if(max(b808d(step(x) + 1) == 9, b808d(step(x) + 1) == 6), 1, 0.85 - ph(x)))

# ======================================================================== drums

struck(s) = max(b808d(s) == 1, b808d(s) == 3)
kickd(s) = if(min(on(drums_on, s), struck(s)) > 0, if(pos(s) == 0, 8, 6), 0)
snared(s) = if(on(drums_on, s) == 0, 0, d16(if(on(snares, s) == 2, 0000000080000303, 0000000080000000), pos(s)))
hit(d, x) = if(d > 0, 0.5 - ph(x), -1)

# Hat rate per beat (hits per beat: 0 none, 2 eighths, 4 sixteenths, 6 triplets, 8 thirty-seconds),
# sixteen beats per number, one number per four bars.
hatgroup(x) = mod(floor(x / 16), 4)
hatrate(x) = d16(if(hatgroup(x) < 1, 68, if(hatgroup(x) < 2, 4444444444444486, if(hatgroup(x) < 3, 4444444444444448, 2222222244444466))), mod(floor(x), 16))
roll(x) = hatrate(x) > 4
hatvel(x) = if(roll(x), 0.3 + 0.6 frac(x), 0.45 + 0.35 (mod(floor(4x), 2) == 0) + 0.1 floor(3 frac(sin(12.9898 floor(4x)) * 43758.5453)) - 0.1)

# ======================================================================== the bell motif

# Codes: 0 rest, 9 hold, 1 E5, 2 F#5, 3 G5, 4 A5, 5 B5, 6 B4, 7 E6, 8 D#5.
bdeg(c) = if(c == 6, -3, if(c == 7, 7, if(c == 8, -1, c - 1)))
# A: B G F# E D# E on a 3-3-2 / 3-3-2 grid.  B: E held, G, F#, down to B4.  B': the same, rising to E6.
belld(s) = d16(if(mod(bar(s), 2) == 0, 5993992919989919, if(mod(bar(s), 8) == 7, 1999993949997999, 1999993929996999)), pos(s))
bheld(s) = (belld(s) == 9)(1 + (belld(s-1) == 9)(1 + (belld(s-2) == 9)(1 + (belld(s-3) == 9)(1 + (belld(s-4) == 9)(1 + (belld(s-5) == 9))))))
bellnote(x) = hm(bdeg(belld(step(x) - bheld(step(x)))))
bellgate(x) = if(belld(step(x)) == 0, -1, if(belld(step(x) + 1) == 9, 1, 0.9 - ph(x)))
# The intro opens the bell's filter from dull to bright over four bars.
bellcut(x) = if(x < 16, 500 * 2^(3 x / 16), 4000)

# A sidechain read from the kick pattern: sixteenths since the last kick.
kage(s) = if(kickd(s) > 0, 0, if(kickd(s - 1) > 0, 1, if(kickd(s - 2) > 0, 2, if(kickd(s - 3) > 0, 3, 4))))
duck(x) = 1 - 0.6 smoothstep(clamp((kage(step(x)) + ph(x)) / 0.15, 0, 1)) (1 - smoothstep(clamp((kage(step(x)) + ph(x) - 0.15) / 3, 0, 1)))

# ======================================================================== voices

voice b808 {
  wave = sine
  pitch = key - 36 + p808(t)
  gate = gate808(t)
  gain = 0.42
  attack = 1
  decay = 1400
  sustain = 0.25
  release = 60
  drive = 3
}

drum kick {
  kind = kick
  trigger = hit(kickd(step(t)), t)
  amp = kickd(step(t)) / 8
  gain = 0.75
}

drum snare {
  kind = snare
  trigger = hit(snared(step(t)), t)
  amp = snared(step(t)) / 8
  gain = 0.7
}

drum hat {
  kind = hat
  trigger = if(min(hatrate(t), on(hats_on, step(t))) > 0, 0.5 - frac(hatrate(t) t), -1)
  amp = hatvel(t)
  gain = 0.28
  pan = 0.25
}

voice bell {
  wave = triangle
  pitch = key + 12 + bellnote(t)
  gate = bellgate(t)
  gain = 0.36
  pan = 0.15
  attack = 1
  decay = 500
  sustain = 0.15
  release = 300
  cutoff = bellcut(t)
}

# The bell an octave down and a dotted eighth late: a dark echo, as a function of time.
voice bell_echo {
  wave = sine
  pitch = key + bellnote(t - 0.75)
  gate = bellgate(t - 0.75)
  gain = 0.14
  pan = -0.5
  attack = 2
  decay = 600
  sustain = 0.1
  release = 400
}

# A low, dark triad pad; re-struck at each chord, ducking under the kick.
padgate(x) = if(frac(x / 8) > 0.97, -1, 1)

voice pad1 {
  wave = saw
  pitch = key - 12 + tri(0, step(t), -2)
  gate = padgate(t)
  amp = duck(t)
  gain = 0.05
  pan = -0.4
  attack = 400
  decay = 3000
  sustain = 0.5
  release = 300
  cutoff = 500
  resonance = 0.3
}

voice pad2 {
  wave = saw
  pitch = key - 12 + tri(1, step(t), -2) + 0.07
  gate = padgate(t)
  amp = duck(t)
  gain = 0.045
  attack = 400
  decay = 3000
  sustain = 0.5
  release = 300
  cutoff = 500
  resonance = 0.3
}

voice pad3 {
  wave = saw
  pitch = key - 12 + tri(2, step(t), -2) - 0.07
  gate = padgate(t)
  amp = duck(t)
  gain = 0.045
  pan = 0.4
  attack = 400
  decay = 3000
  sustain = 0.5
  release = 300
  cutoff = 500

}
