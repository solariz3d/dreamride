# City-pop sketch: our own short loop, written the way graphing-calculator music is written.
# Eight bars of 4/4 at 100 bpm. Pitch is in semitones from A4 = 440 Hz.

bpm 100
beats 32
rate 48000
reference 440

# ---- the reference formulas (DESIGN.md) as functions ----

# Scale degree to major-scale semitone. DESIGN.md writes +2, which gives the natural minor
# scale (0 2 3 5 7 8 10); +5 gives major (0 2 4 5 7 9 11). This sketch is in major.
h(d) = floor((12d + 5) / 7)

# Chord roots stored as the digits of one integer, minus 5: -2 -1 -3 -3 0 3 -1 2 -2.
root(i) = mod(floor(342258473 / 10^(8 - i)), 10) - 5

# Keeps a voicing in [-9, 2] semitones.
fold(z) = mod(z + 9, 12) - 9

# Swing as a time warp: within each half beat, the second sixteenth lands late.
sw(x) = (floor(2x) + max(frac(2x) / 1.16, (frac(2x) - 0.16) / 0.84)) / 2

# Smoothstep, for chord glides.
smoothstep(u) = u^2(3 - 2u)

# ---- the song ----

key = 3                                  # C5, three semitones above A4
bar(x) = floor(x / 4)
deg(x) = root(mod(bar(x), 8))            # this bar's chord root, as a scale degree

# Chord tone k (0 root, 1 third, 2 fifth, 3 seventh) of the bar at time x.
tone(k, x) = fold(h(deg(x) + 2k))
# Glide into the next bar's chord over the last quarter beat of each bar.
glide(x) = smoothstep(clamp((frac(x / 4) * 4 - 3.75) / 0.25, 0, 1))
chord(k, x) = key + tone(k, x) + (tone(k, x + 4) - tone(k, x)) glide(x)

# A soft pump on every beat, as if side-chained to the kick.
pump(x) = 0.35 + 0.65 smoothstep(clamp(frac(x) * 2.5, 0, 1))

# Lead: one 16-digit integer per bar (two alternating), one digit per swung sixteenth.
# Digits are scale degrees above the chord root; 9 is a rest.
digit16(n, i) = mod(floor(n / 10^(15 - i)), 10)
melody(x) = 4024642099402460 + (6424202199624757 - 4024642099402460) mod(bar(x), 2)
step(x) = mod(floor(4 sw(x)), 16)
note(x) = digit16(melody(x), step(x))

voice pad_root {
  wave = triangle
  pitch = chord(0, t)
  gate = 1
  amp = pump(t)
  gain = 0.16
  pan = -0.35
}

voice pad_third {
  wave = triangle
  pitch = chord(1, t)
  gate = 1
  amp = pump(t)
  gain = 0.14
  pan = 0.35
}

voice pad_fifth {
  wave = triangle
  pitch = chord(2, t)
  gate = 1
  amp = pump(t)
  gain = 0.14
  pan = -0.15
}

voice pad_seventh {
  wave = triangle
  pitch = chord(3, t)
  gate = 1
  amp = pump(t)
  gain = 0.12
  pan = 0.15
}

# Disco octave bass on the roots: eighth notes, every second one up an octave.
voice bass {
  wave = saw
  pitch = key - 24 + fold(h(deg(t))) + 12 mod(floor(2t), 2)
  gate = 0.7 - frac(2t)
  gain = 0.32
  release = 20
}

voice lead {
  wave = square
  pitch = key + h(note(t) + deg(t))
  gate = min(8.5 - note(t), 0.75 - frac(4 sw(t)))
  gain = 0.11
  pan = 0.2
}

# Harmony a third (two scale degrees) above the lead.
voice harmony {
  wave = triangle
  pitch = key + h(note(t) + deg(t) + 2)
  gate = min(8.5 - note(t), 0.75 - frac(4 sw(t)))
  gain = 0.09
  pan = -0.2
}

drum kick {
  kind = kick
  trigger = 0.25 - frac(t)
  gain = 0.9
}

drum snare {
  kind = snare
  trigger = min(mod(floor(t), 2) - 0.5, 0.25 - frac(t))
  gain = 0.5
  pan = 0.05
}

drum hat {
  kind = hat
  trigger = 0.5 - frac(4 sw(t))
  amp = 0.55 + 0.45 mod(floor(4 sw(t)) + 1, 2)
  gain = 0.35
  pan = 0.3
}
